# #2.Реализация программы
```c
#include <stdio.h>
#include <locale.h>

int main() {
    setlocale(LC_ALL, "Russian");

    int N;
    double P = 1.0;

    printf("Введите целое число N: ");
    if (scanf("%d", &N) != 1 || N < 1) {
        printf("Ошибка: N должно быть целым положительным числом.\n");
        return 1;
    }

    for (int i = 1; i <= N; i++) {
        P *= (1.0 + i / 10.0);
    }

    printf("Произведение равно: %.6f\n", P);

    return 0;
}
```
